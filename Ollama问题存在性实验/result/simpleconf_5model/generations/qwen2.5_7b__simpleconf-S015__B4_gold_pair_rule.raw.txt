def nested_with_profiles(profiles):
    config = ProfileConfig.load(profiles, base='default')
    
    with ProfileConfig.with_profile(config, 'prod'):
        a = config.x
        
    with ProfileConfig.with_profile(config, 'default'):
        b = config.x
    
    c = config.x
    
    return (a, b, c)
