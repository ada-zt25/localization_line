def get_beta_wrapper_disable_then_score(value):
    with sp.plugins_context([beta]):
        beta_wrapper = sp.get_plugin(beta)
        beta_wrapper.disable()
        result = sp.hooks.score(value)
        beta_wrapper.enable()
    return result
