def beta_only_inside_context_then_all_score(value):
    with sp.plugins_context(plugins=[beta]):
        result_with_beta = sp.hooks.score(value)
    
    original_plugins = [sp.get_plugin(name) for name in sp.enabled_plugins]
    for plugin in original_plugins:
        plugin.enable()
    
    result_with_original = sp.hooks.score(value)
    
    return result_with_beta, result_with_original
