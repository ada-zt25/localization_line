def to_plain_dict(data):
    if isinstance(data, dict):
        diot_data = Diot(data)
        return {k: to_plain_dict(v) for k, v in diot_data.items()}
    elif isinstance(data, (Diot, OrderedDiot, FrozenDiot)):
        try:
            data.thaw()
            return to_plain_dict(data.to_dict())
        except DiotFrozenError:
            return data.to_dict()
    else:
        return data
