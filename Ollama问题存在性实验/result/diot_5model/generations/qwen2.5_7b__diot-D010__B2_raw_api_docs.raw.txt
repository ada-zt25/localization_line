def nested_mutate_reflects(data, val):
    from diot import Diot
    
    # Convert input data to a Diot with nested dict values converted to Diots
    d = Diot(data, diot_nest=True)
    
    # Set the value of d.x.y
    d['x']['y'] = val
    
    # Read the value back from the same Diot
    result = d.get('x', {}).get('y', None)
    
    return result
