def persistent_disable_beta_then_temp_only_alpha_score(value):
    sp, alpha, beta, gamma = make_score_manager()
    
    # Disable beta by name
    sp.disable('beta')
    
    # Collect score with only alpha enabled in a temporary context
    with plugins_context([alpha]):
        inside_result = sp.hooks.score(value)
    
    # Re-enable beta after the context exits
    sp.enable('beta')
    
    # Collect score again to see the effect of disabling beta
    outside_result = sp.hooks.score(value)
    
    return inside_result, outside_result
