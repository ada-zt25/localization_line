def remap(pairs, key, val):
    b = bidict(pairs)
    try:
        b.forceput(key, val)
    except ValueDuplicationError:
        old_key = next(k for k, v in pairs.items() if v == val and k != key)
        b.forceput(old_key, pairs[old_key])
        del pairs[old_key]
        b.forceput(old_key, val)
    return dict(b)
