def disable_alpha_gamma_then_score(value):
    sp, alpha, beta, gamma = make_score_manager()
    with sp.plugins_context([beta, gamma]):
        beta_wrapper = sp.get_plugin(beta)
        gamma_wrapper = sp.get_plugin(gamma)
        beta_wrapper.disable()
        gamma_wrapper.disable()
        result = sp.hooks.score(value)
    return result
