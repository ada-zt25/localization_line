def beta_only_inside_context_then_all_score(value):
    sp, alpha, beta, gamma = make_score_manager()

    with sp.plugins_context([beta]):
        sp.disable(alpha)
        sp.enable(beta)
        result1 = sp.hooks.score(value)

    sp.enable(alpha)
    sp.disable(beta)
    result2 = sp.hooks.score(value)

    return result1, result2
