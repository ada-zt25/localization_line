def disable_alpha_gamma_via_wrappers_then_score(value):
    sp, alpha, beta, gamma = make_score_manager()

    with plugins_context([alpha.name, gamma.name]):
        alpha_wrapper = get_plugin(alpha.name)
        alpha_wrapper.disable()
        gamma_wrapper = get_plugin(gamma.name)
        gamma_wrapper.disable()

    result = sp.hooks.score(value)
    return result
