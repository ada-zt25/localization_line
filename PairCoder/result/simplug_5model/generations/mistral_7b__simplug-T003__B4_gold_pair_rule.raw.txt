def disable_beta_then_score(value):
    sp, alpha, beta, gamma = make_score_manager()
    with sp.plugins_context([alpha, gamma]):
        beta_wrapper = sp.get_plugin(beta)
        beta_wrapper.disable()
        result = sp.hooks.score(value)
    return result
