from simpleconf import Config, ProfileConfig

def with_profile_keeps_current(profiles):
    config = Config()
    profile_config = ProfileConfig(config, profiles)
    
    # Load and set to 'prod'
    profile_config.load('prod')
    
    with profile_config.with_profile('default'):
        active_profile = profile_config.get_active_profile()
    
    return profile_config.get_active_profile()
