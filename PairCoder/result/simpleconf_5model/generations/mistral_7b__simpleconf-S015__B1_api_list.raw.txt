from simpleconf import Config, ProfileConfig

def nested_with_profiles(profiles):
    with ProfileConfig.load(profiles, base='default').use_profile('prod'):
        with ProfileConfig.load(profiles, base='default').with_profile('default'):
            a = Config.get('x')
        c = Config.get('x')
        b = Config.get('x')  # 'default' profile is still active here
    return a, b, c
