def switch_two_profiles(profiles):
    prod_config = profiles['prod']
    default_config = profiles['default']

    prod_config.load()
    prod_x = prod_config.get('x')

    default_config.load()
    default_config.set('x', prod_x)
    default_config.save()

    default_config.load()
    default_x = default_config.get('x')

    return (prod_x, default_x)
