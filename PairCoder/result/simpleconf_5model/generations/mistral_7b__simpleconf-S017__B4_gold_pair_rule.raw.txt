import simpleconf

def double_use_persistent(profiles):
    conf = simpleconf.ProfileConfig.load(profiles, base='default')
    current_profile = conf.current_profile(conf)

    with conf.with_profile(conf, 'prod'):
        conf.load(*profiles, loader=None)  # reload with persistently switched profile
        _ = conf.x  # ignore the value, we just need to load it to switch profiles

    conf.use_profile(conf, current_profile)  # switch back to the original profile

    return conf.x  # return the final, persistent value of x in the 'default' profile
