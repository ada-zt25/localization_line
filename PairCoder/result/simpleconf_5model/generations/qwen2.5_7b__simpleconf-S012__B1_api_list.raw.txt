```python
from simpleconf import Config, ProfileConfig

def with_profile_temporary(profiles, profile):
    conf = Config.load(*profiles)
    original_profile = ProfileConfig.current_profile(conf)
    
    inside = None
    after = None
    
    with ProfileConfig.with_profile(conf, profile):
        inside = conf.x
    
    ProfileConfig.use_profile(conf, original_profile)
    after = conf.x
    
    return (inside, after)
```
