```python
from diot import Diot, OrderedDiot, FrozenDiot

def nested_chain(data):
    d = Diot()
    for key, value in data.items():
        if isinstance(value, dict):
            d[key] = OrderedDiot(sorted(value.items()))
        elif isinstance(value, FrozenDiot):
            d[key] = value
        else:
            d[key] = value
    return d.x.y
```
